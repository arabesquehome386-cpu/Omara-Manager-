const SHEET_NAME = 'OmaraBackup'; const PAYS_SHEET = 'سجل المدفوعات'; const CHUNK_SIZE = 40000; function doPost(e) { try { const data = JSON.parse(e.postData.contents); // ---- UPLOAD FILE TO DRIVE ---- if(data.type === 'upload') { const bytes = Utilities.base64Decode(data.fileData); const blob = Utilities.newBlob(bytes, data.mimeType, data.fileName); const folder = getOrCreateFolder(data.folderName || 'OmaraReceipts'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const url = 'https://drive.google.com/file/d/' + file.getId() + '/view'; return ContentService.createTextOutput(JSON.stringify({ok:true, url:url})) .setMimeType(ContentService.MimeType.JSON); } // ---- BACKUP JSON DATA ---- const jsonStr = JSON.stringify(data.payload); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (sh) { sh.clear(); } else { sh = ss.insertSheet(SHEET_NAME); } sh.getRange(1, 1, 2000, 1).setNumberFormat('@STRING@'); const chunks = []; for (let i = 0; i < jsonStr.length; i += CHUNK_SIZE) chunks.push(jsonStr.substring(i, i + CHUNK_SIZE)); sh.getRange(1, 1).setValue(new Date().toISOString()); sh.getRange(1, 2).setValue(chunks.length); chunks.forEach((c, idx) => sh.getRange(idx + 2, 1).setValue(c)); // ---- WRITE HUMAN-READABLE PAYMENTS SHEET ---- try { let psh = ss.getSheetByName(PAYS_SHEET); if (!psh) psh = ss.insertSheet(PAYS_SHEET); psh.clearContents(); psh.getRange(1,1,1,8).setValues([['الساكن','التاريخ','المبلغ','الطريقة','رابط الإيصال','البيان','توزيع على الشهور','المتأخرات اليدوية']]); const rows = []; if(data.payload && data.payload.sokan) { data.payload.sokan.forEach(r => { (r.payments||[]).forEach(p => { const breakdown = (p.allocationBreakdown||[]).map(b=>b.desc+': '+b.covered+' ج.م').join(' | '); const manualDebts = (r.manualDebts||[]).map(d=>d.desc+': '+d.amount+' ج.م').join(' | '); rows.push([r.name||'—', p.date||'', p.amount||0, p.method||'', p.evidenceLink||'', p.note||'', breakdown, manualDebts]); }); }); } if(rows.length) psh.getRange(2,1,rows.length,8).setValues(rows); } catch(e2) {} // ---- WRITE MAINTENANCE INVOICES (FAWATIR) SHEET ---- try { const FAW_SHEET = 'فواتير الصيانة'; let fsh = ss.getSheetByName(FAW_SHEET); if (!fsh) fsh = ss.insertSheet(FAW_SHEET); fsh.clearContents(); fsh.getRange(1,1,1,5).setValues([['العنوان','التاريخ','الشهر المرتبط','رابط Drive','عدد الصور المرفقة']]); const frows = []; if(data.payload && data.payload.fawatir) { const months = ['','يناير','فبراير','مارس','أبريل','مايو','يونيو','يوليو','أغسطس','سبتمبر','أكتوبر','نوفمبر','ديسمبر']; data.payload.fawatir.forEach(f => { const monthLabel = f.month !== undefined && data.payload.months && data.payload.months[parseInt(f.month)] ? months[data.payload.months[parseInt(f.month)].month]+' '+data.payload.months[parseInt(f.month)].year : ''; frows.push([f.title||'—', f.date||'', monthLabel, f.driveLink||'', (f.images||[]).length]); }); } if(frows.length) fsh.getRange(2,1,frows.length,5).setValues(frows); } catch(e3) {} return ContentService.createTextOutput(JSON.stringify({ok:true, chunks:chunks.length})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok:false, err:err.message})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sh = ss.getSheetByName(SHEET_NAME); if (!sh) return ContentService.createTextOutput(JSON.stringify({ok:false,err:'لا توجد نسخة محفوظة بعد'})).setMimeType(ContentService.MimeType.JSON); const savedAt = sh.getRange(1,1).getValue().toString(); const chunkCount = parseInt(sh.getRange(1,2).getValue()) || 0; if (!chunkCount) return ContentService.createTextOutput(JSON.stringify({ok:false,err:'لا توجد نسخة محفوظة بعد'})).setMimeType(ContentService.MimeType.JSON); let full = ''; for (let i = 0; i < chunkCount; i++) full += sh.getRange(i+2,1).getValue().toString(); return ContentService.createTextOutput(JSON.stringify({ok:true, data:full, savedAt:savedAt})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok:false,err:err.message})).setMimeType(ContentService.MimeType.JSON); } } function getOrCreateFolder(name) { const folders = DriveApp.getFoldersByName(name); return folders.hasNext() ? folders.next() : DriveApp.createFolder(name); }
